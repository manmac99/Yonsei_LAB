        -m 2G \
        -smp 2 \

        -net user,host=10.0.2.10,hostfwd=tcp:127.0.0.1:10021-:22 \

새로운 run.sh의 내용{
#!/bin/bash

KERNEL_PATH="/home/poby/Desktop/linux/arch/x86/boot/bzImage"
DISK_IMAGE="bullseye.img"

qemu-system-x86_64 \
    -m 2G \
    -smp 2 \
    -kernel "$KERNEL_PATH" \
    -append "console=ttyS0 root=/dev/sda earlyprintk=serial net.ifnames=0 nokaslr" \
    -drive file="$DISK_IMAGE",format=raw \
    -net user,host=10.0.2.10,hostfwd=tcp:127.0.0.1:10021-:22 \
    -net nic,model=e1000 \
    -enable-kvm \
    -nographic \
    -pidfile vm.pid \
    2>&1 | tee vm.log

}
